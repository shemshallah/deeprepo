cmake_minimum_required(VERSION 3.16)
project(repovram_fuse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
add_executable(repovram-fuse main.cpp)
target_link_libraries(repovram-fuse PRIVATE PkgConfig::FUSE3 PkgConfig::CURL pthread)
target_compile_options(repovram-fuse PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
