cmake_minimum_required(VERSION 3.16)
project(ram16_gguf_fuse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(ram16-gguf-mount main.cpp)
target_link_libraries(ram16-gguf-mount PRIVATE PkgConfig::FUSE3 PkgConfig::CURL OpenSSL::Crypto ZLIB::ZLIB)
target_compile_options(ram16-gguf-mount PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
install(TARGETS ram16-gguf-mount RUNTIME DESTINATION bin)
